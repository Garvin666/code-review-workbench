Guardian 上报片段 · curl 定时任务
=====================================
把 token 替换为你在工作台登记目标后拿到的明文 token；把 hub 替换为实际地址。

# 心跳（推荐加入 crontab 每 60s 一次，或 GitHub Actions、Azure Pipelines 等 CI 步）
curl -s -X POST http://127.0.0.1:8700/api/v1/report \
  -H "Content-Type: application/json" \
  -H "X-Guardian-Token: <token>" \
  -d '{"metric":"online","ok":true}'

# 性能上报（value = 客户端测得的请求耗时）
curl -s -X POST http://127.0.0.1:8700/api/v1/report \
  -H "Content-Type: application/json" \
  -H "X-Guardian-Token: <token>" \
  -d '{"metric":"latency","ok":true,"value":3200}'

# 一次性异常事件
curl -s -X POST http://127.0.0.1:8700/api/v1/event \
  -H "Content-Type: application/json" \
  -H "X-Guardian-Token: <token>" \
  -d '{"type":"crash","message":"OOM in worker-3"}'

# 一键登记新目标（管理用，返回明文 token，仅此一次）
curl -s -X POST http://127.0.0.1:8700/api/v1/targets \
  -H "Content-Type: application/json" \
  -d '{"name":"我的商城","url":"https://shop.example.com","type":"website"}'

# ================= v0.3 管理 / 自测示例 =================
# 登记前连通性自测（单次探测，不入库不改状态）
curl -s -X POST http://127.0.0.1:8700/api/v1/validate \
  -H "Content-Type: application/json" \
  -d '{"url":"https://shop.example.com","probes":["online","latency","tls","content","security"]}'

# 轮换 token（旧 token 立即失效；响应中的新 token 仅展示一次）
curl -s -X POST http://127.0.0.1:8700/api/v1/targets/<target-id>/token/rotate \
  -H "Content-Type: application/json" -d '{}'

# 暂停守护 / 恢复 / 改参数（enabled 布尔开关，interval_s 秒）
curl -s -X POST http://127.0.0.1:8700/api/v1/targets/<target-id> \
  -H "Content-Type: application/json" \
  -d '{"enabled":false}'

# 注销目标（历史事件保留）
curl -s -X DELETE http://127.0.0.1:8700/api/v1/targets/<target-id>

# 通知渠道：保存（预设 webhook|qyweixin|dingtalk|feishu|serverchan）
curl -s -X POST http://127.0.0.1:8700/api/v1/notify \
  -H "Content-Type: application/json" \
  -d '{"action":"save","enabled":true,"preset":"qyweixin","url":"https://qyapi.weixin.qq.com/cgi-bin/webhook/send?key=xxx","levels":["warn","critical"]}'

# 通知渠道：发送测试
curl -s -X POST http://127.0.0.1:8700/api/v1/notify \
  -H "Content-Type: application/json" -d '{"action":"test"}'

# 维护窗口：新增（start/end 用 ISO 时间；targets 留空 = 全部目标）
curl -s -X POST http://127.0.0.1:8700/api/v1/maintenance \
  -H "Content-Type: application/json" \
  -d '{"action":"add","title":"周三例行发布","start":"2026-09-09T02:00:00+08:00","end":"2026-09-09T03:00:00+08:00","targets":[]}'

# 维护窗口：删除
curl -s -X POST http://127.0.0.1:8700/api/v1/maintenance \
  -H "Content-Type: application/json" \
  -d '{"action":"delete","id":"mt-xxx"}'
